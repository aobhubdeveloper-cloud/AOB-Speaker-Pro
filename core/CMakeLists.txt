cmake_minimum_required(VERSION 3.24)
project(aob_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(aob_protocol STATIC protocol/src/packet.cpp)
target_include_directories(aob_protocol PUBLIC protocol/include)
add_library(aob_dsp STATIC dsp/src/dsp.cpp)
target_include_directories(aob_dsp PUBLIC dsp/include)
enable_testing()
add_executable(aob_protocol_test protocol/tests/packet_test.cpp)
target_link_libraries(aob_protocol_test PRIVATE aob_protocol)
add_test(NAME protocol COMMAND aob_protocol_test)
add_executable(aob_dsp_test dsp/tests/dsp_test.cpp)
target_link_libraries(aob_dsp_test PRIVATE aob_dsp)
add_test(NAME dsp COMMAND aob_dsp_test)
