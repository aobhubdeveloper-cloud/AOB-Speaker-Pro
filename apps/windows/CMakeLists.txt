cmake_minimum_required(VERSION 3.24)
project(AOBSpeakerProWindows LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(opus CONFIG REQUIRED)
add_executable(aob-speaker-pro WIN32 src/main.cpp src/wasapi_loopback.cpp src/udp_sender.cpp src/opuspipeline.cpp)
target_include_directories(aob-speaker-pro PRIVATE ../../core/protocol/include ../../core/dsp/include)
target_link_libraries(aob-speaker-pro PRIVATE aob_protocol aob_dsp opus::opus ole32 avrt mmdevapi ws2_32)
if(MSVC)
  target_compile_options(aob-speaker-pro PRIVATE /W4 /permissive-)
endif()
